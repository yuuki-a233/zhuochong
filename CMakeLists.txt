cmake_minimum_required(VERSION 3.16)
project(YuukiDesktopPet VERSION 0.1.2 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(YuukiDesktopPet WIN32 src/main.cpp resources.rc)
target_include_directories(YuukiDesktopPet PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(YuukiDesktopPet PRIVATE gdiplus ole32 shell32 gdi32 user32 advapi32 uuid)
if(MINGW)
 target_link_options(YuukiDesktopPet PRIVATE -municode -static -static-libgcc -static-libstdc++)
endif()
if(MSVC)
 target_compile_options(YuukiDesktopPet PRIVATE /utf-8)
 set_property(TARGET YuukiDesktopPet PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
